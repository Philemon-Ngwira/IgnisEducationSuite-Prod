import { useState, useEffect } from 'react';
import type { Form, FormField, FormSubmission } from '../type';
import { useAppStore } from '../store';
import { v4 as uuidv4 } from 'uuid';
import { AutocompleteInput } from './AutocompleteInput';

interface FormRendererProps {
  form: Form;
  onSuccess?: () => void;
}

interface DynamicOption {
  label: string;
  value: string;
}

export function FormRenderer({ form, onSuccess }: FormRendererProps) {
  const addSubmission = useAppStore((state) => state.addSubmission);

  const [formData, setFormData] = useState<Record<string, unknown>>({});
  const [errors, setErrors] = useState<Record<string, string>>({});
  const [submitted, setSubmitted] = useState(false);
  const [dynamicOptions, setDynamicOptions] = useState<Record<string, DynamicOption[]>>({});
  const [loadingOptions, setLoadingOptions] = useState<Record<string, boolean>>({});

  const fetchOptionsForField = async (field: FormField, searchQuery?: string) => {
    if (!field.dynamicUrl) return;
    setLoadingOptions((prev) => ({ ...prev, [field.id]: true }));

    try {
      let url = field.dynamicUrl;
      if (searchQuery !== undefined && field.dynamicSearchParam) {
        const urlObj = new URL(url);
        urlObj.searchParams.set(field.dynamicSearchParam, searchQuery);
        url = urlObj.toString();
      }

      const response = await fetch(url);
      const data = await response.json();

      const labelKey = field.dynamicLabelKey || 'label';
      const valueKey = field.dynamicValueKey || 'value';

      const optionsData: unknown[] = Array.isArray(data) ? data : [];

      const mapped = optionsData.map((item) => {
        const record = item as Record<string, unknown>;
        return {
          label: String(record[labelKey] ?? item),
          value: String(record[valueKey] ?? item),
        };
      });

      setDynamicOptions((prev) => ({ ...prev, [field.id]: mapped }));
    } catch (err) {
      console.error(`Failed to fetch options for ${field.label}:`, err);
    } finally {
      setLoadingOptions((prev) => ({ ...prev, [field.id]: false }));
    }
  };

  useEffect(() => {
    form.fields.forEach((field) => {
      const needsFetch =
        (field.type === 'dropdown' || field.type === 'autocomplete') &&
        field.dataSource === 'dynamic' &&
        field.dynamicUrl;

      // Dropdown: fetch full list immediately.
      // Autocomplete: only fetch immediately if it has no search param (rare) —
      // otherwise wait for the user to type.
      if (needsFetch && !(field.type === 'autocomplete' && field.dynamicSearchParam)) {
        fetchOptionsForField(field);
      }
    });
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [form.fields]);

  const validate = (): boolean => {
    const newErrors: Record<string, string> = {};

    form.fields.forEach((field) => {
      const value = formData[field.id];
      const isBoolean = field.type === 'checkbox' || field.type === 'switch';

      if (field.required) {
        if (isBoolean && value !== true) {
          newErrors[field.id] = 'This field is required';
        } else if (!isBoolean && (value === undefined || value === null || String(value).trim() === '')) {
          newErrors[field.id] = 'This field is required';
        }
      }
    });

    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const handleChange = (fieldId: string, value: unknown) => {
    setFormData((prev) => ({ ...prev, [fieldId]: value }));
    setErrors((prev) => {
      if (!prev[fieldId]) return prev;
      const next = { ...prev };
      delete next[fieldId];
      return next;
    });
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!validate()) return;

    const submission: FormSubmission = {
      id: uuidv4(),
      formId: form.id,
      data: formData,
      submittedAt: new Date().toISOString(),
    };

    try {
      await addSubmission(submission);
      setSubmitted(true);
      onSuccess?.();
    } catch (err) {
      console.error('Submission failed:', err);
      setErrors((prev) => ({ ...prev, __submit: 'Failed to submit. Please try again.' }));
    }
  };

  if (submitted) {
    return (
      <div style={{ maxWidth: 600, margin: '2rem auto', textAlign: 'center' }}>
        <h2>Successfully Submitted</h2>
        <p>Your response has been recorded.</p>
        <button
          onClick={() => {
            setFormData({});
            setSubmitted(false);
          }}
        >
          Submit another response
        </button>
      </div>
    );
  }

  return (
    <div style={{ maxWidth: 700, margin: '2rem auto' }}>
      <h2>{form.title}</h2>

      <form onSubmit={handleSubmit}>
        <div
          style={{
            display: 'grid',
            gridTemplateColumns: `repeat(${form.gridColumns}, 1fr)`,
            gap: '1rem',
          }}
        >
          {form.fields.map((field) => {
            const hasError = !!errors[field.id];

            return (
              <div
                key={field.id}
                style={{
                  gridColumn: `${field.col + 1} / span ${field.colSpan}`,
                  gridRow: `${field.row + 1} / span ${field.rowSpan}`,
                }}
              >
                <label style={{ display: 'block', fontSize: '0.8rem', fontWeight: 600, marginBottom: 4 }}>
                  {field.label}
                  {field.required && <span style={{ color: 'blue' }}> *</span>}
                </label>

                {field.type === 'text' && (
                  <input
                    type="text"
                    value={(formData[field.id] as string) || ''}
                    onChange={(e) => handleChange(field.id, e.target.value)}
                  />
                )}

                {field.type === 'number' && (
                  <input
                    type="number"
                    value={(formData[field.id] as string) || ''}
                    onChange={(e) => handleChange(field.id, e.target.value)}
                  />
                )}

                {field.type === 'date' && (
                  <input
                    type="date"
                    value={(formData[field.id] as string) || ''}
                    onChange={(e) => handleChange(field.id, e.target.value)}
                  />
                )}

                {field.type === 'dropdown' && (
                  <select
                    value={(formData[field.id] as string) || ''}
                    onChange={(e) => handleChange(field.id, e.target.value)}
                    disabled={loadingOptions[field.id]}
                  >
                    <option value="" disabled>
                      {loadingOptions[field.id] ? 'Loading...' : 'Select an option...'}
                    </option>
                    {field.dataSource === 'dynamic'
                      ? dynamicOptions[field.id]?.map((opt) => (
                          <option key={opt.value} value={opt.value}>
                            {opt.label}
                          </option>
                        ))
                      : field.options?.map((opt) => (
                          <option key={opt} value={opt}>
                            {opt}
                          </option>
                        ))}
                  </select>
                )}

                {field.type === 'autocomplete' && (
                  <AutocompleteInput
                    options={
                      field.dataSource === 'dynamic'
                        ? dynamicOptions[field.id] || []
                        : field.options?.map((opt) => ({ label: opt, value: opt })) || []
                    }
                    value={(formData[field.id] as string) || ''}
                    onChange={(val) => handleChange(field.id, val)}
                    isLoading={loadingOptions[field.id]}
                    onSearch={
                      field.dataSource === 'dynamic' && field.dynamicSearchParam
                        ? (query) => fetchOptionsForField(field, query)
                        : undefined
                    }
                  />
                )}

                {field.type === 'checkbox' && (
                  <input
                    type="checkbox"
                    checked={!!formData[field.id]}
                    onChange={(e) => handleChange(field.id, e.target.checked)}
                  />
                )}

                {field.type === 'switch' && (
                  <input
                    type="checkbox"
                    role="switch"
                    checked={!!formData[field.id]}
                    onChange={(e) => handleChange(field.id, e.target.checked)}
                  />
                )}

                {hasError && (
                  <p style={{ color: 'red', fontSize: '0.75rem', marginTop: 4 }}>{errors[field.id]}</p>
                )}
              </div>
            );
          })}
        </div>

        {errors.__submit && <p style={{ color: 'red' }}>{errors.__submit}</p>}

        <button type="submit" style={{ marginTop: '1.5rem' }}>
          {form.submitLabel || 'Submit Response'}
        </button>
      </form>
    </div>
  );
}
